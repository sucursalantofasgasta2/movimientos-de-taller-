<script>
    // ******************************************************
    // TU CONEXIÓN CON GOOGLE SHEETS
    // ******************************************************
    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzDztjKx7v6syzE0V3sAcufP6v_zTVLZs2QczgniswhkjtyKIiO6OQ_p9Fv9lzHH-HN/exec"; 

    // Poner fecha de hoy automáticamente
    document.getElementById('fecha').valueAsDate = new Date();

    // Función simple para contar unidades por espacios o comas
    function contar(texto) {
        if (!texto || texto.trim() === '') return 0;
        return texto.replace(/,/g, ' ').replace(/\n/g, ' ').trim().split(/\s+/).filter(i => i.length > 0).length;
    }

    const form = document.getElementById('tallerForm');
    const loading = document.getElementById('loading');

    form.addEventListener('submit', e => {
        e.preventDefault();
        
        // Mostrar cargando
        loading.style.display = 'flex';

        // Recopilar datos del formulario
        const ingresosTxt = document.getElementById('ingresos').value;
        const pendientesTxt = document.getElementById('pendientes').value;
        const retirosTxt = document.getElementById('retiros').value;

        // Crear el paquete de datos
        const data = {
            fecha: document.getElementById('fecha').value,
            ingresos: ingresosTxt,
            cantIngresos: contar(ingresosTxt),
            pendientes: pendientesTxt,
            cantPendientes: contar(pendientesTxt),
            retiros: retirosTxt,
            cantRetiros: contar(retirosTxt),
            rechazos: document.getElementById('rechazos').value,
            entregas: document.getElementById('entregas').value
        };

        // Enviar a Google Sheets
        fetch(SCRIPT_URL, {
            method: 'POST',
            mode: 'no-cors', 
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(data)
        })
        .then(() => {
            // Cuando termine...
            loading.style.display = 'none';
            alert("✅ ¡Reporte guardado en la Hoja de Cálculo!");
            form.reset(); // Limpiar formulario
            document.getElementById('fecha').valueAsDate = new Date(); // Resetear fecha
        })
        .catch(error => {
            loading.style.display = 'none';
            alert("❌ Hubo un error. Revisa tu conexión a internet.");
            console.error('Error:', error);
        });
    });
</script>
